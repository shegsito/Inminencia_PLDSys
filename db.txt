-- tabla de usuarios
CREATE TABLE usuario (
	IDUSUARIO   	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	nombre 			VARCHAR(200) NOT NULL,
	email 		    VARCHAR(200) NOT NULL UNIQUE,
	password_hash 	VARCHAR(255) NOT NULL,
	rol				VARCHAR(20) NOT NULL CHECK (rol IN ('oficial', 'operador', 'auditor')),
	activo 			BOOLEAN NOT NULL DEFAULT true,
	created_at		TIMESTAMPTZ NOT NULL DEFAULT now()
	-- ..TZ es time zone
);

-- tabla clientes

-- ¿QUE PASO CON LOS NOT NULL?
-- hay campos que no cuentan con el not null por la razón de los clientes registrados,
-- sería imposible registrar datos hasta obtener algunos de los registros como curp hasta obtener ese dato
-- de igual forma hay cleintes que no tienen domicilio físical en méxico o curp mexicana

-- ¿QUE PASO CON LOS ENUM's?
-- Si el cliente SVA LAW quisiera agregar un nuevo parámetro o estatus (imaginemos un nuevo rol), tendríamos
-- que bloquear temporalmente la tabla completa para cambiarlo. Por esto aquí usaremos VARCHAR + CHECK CONSTRAINT
-- permitiendonos simplemente modificar la restricción sin tocar la estructura de la tabla.

CREATE TABLE clientes(
	IDCliente		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	nombre_cliente	VARCHAR(255) NOT NULL,
	RFC				VARCHAR(13) NOT NULL UNIQUE,
	curp			VARCHAR(18) ,
	domicilio 		TEXT ,
	email			VARCHAR(200) NOT NULL UNIQUE,
	telefono		VARCHAR(20) ,

	-- aquí estamos creando lo que decía de los enums, pero con diferentes estados para elegir
	tipo_persona 	VARCHAR(10) NOT NULL CHECK (tipo_persona IN ('fisica' , 'moral')),
	estatus			VARCHAR(10) NOT NULL DEFAULT 'prospecto' CHECK (estatus IN ('prospecto', 'activo', 'suspendido')),
	nivel_riesgo  	VARCHAR(10) NOT NULL DEFAULT 'bajo' CHECK (nivel_riesgo IN ('bajo' , 'medio' , 'alto')),

	bloquedo		BOOLEAN NOT NULL DEFAULT false,
	created_at 		TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- tabla expediente 1 a 1 con cliente
CREATE TABLE expediente(
	IDExpediente	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDCliente		UUID NOT NULL UNIQUE REFERENCES cliente(IDCliente),
	validado_por	UUID REFERENCES usuario(id),
	completo		BOOLEAN NOT NULL DEFAULT false,
	activo			BOOLEAN NOT NULL DEFAULT false,
	fecha_validac	TIMESTAMPTZ,
	created_at 		TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE documento(
	IDDocumento 	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDExpediente	UUID NOT NULL REFERENCES expediente(IDExpendiente),
	cargado_por		UUID REFERENCES usuario(IDUsuario),
	tipo_documento	VARCHAR(50) NOT NULL,
	ruta_archivo	VARCHAR(500) NOT NULL,
	vigente 		BOOLEAN NOT NULL DEFAULT true,
	fecha_carga 	TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Historial de modificaciones al expediente
CREATE TABLE modificacion_expediente (
	IDModificacion	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDExpediente	UUID NOT NULL REFERENCES expendiente(IDExpendiente),
	usuario_id		UUID NOT NULL REFERENCES usuario(id),
	campo_modif		VARCHAR(100) NOT NULL,
	valor_anterior	TEXT,
	valor_nuevo		TEXT,
	justifacion		TEXT NOT NULL,
	fecha			TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Contratos
CREATE TABLE contrato(
	IDContrato		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDCliente		UUID NOT NULL REFERENCES cliente(IDCliente),
	tipo_producto	VARCHAR(50) NOT NULL,
	monto			NUMERIC(15,2) NOT NULL,
	fecha_inicio	DATE NOT NULL,
	fecha_fin		DATE NOT NULL,
	estatus 		VARCHAR(20) NOT NULL DEFAULT 'vigente' CHECK (estatus IN ('vigente', 'vencido', 'cancelado', 'en_revision'))
);

-- Operaciones financieras
CREATE TABLE operacion(
	IDOperacion		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDCliente		UUID NOT NULL REFERENCES cliente(IDCliente),
	IDContrato		UUID NOT NULL REFERENCES contrato(IDContrato),
	registrada_por	UUID REFERENCES usuario(IDusuario),
	tipo_producto 	VARCHAR(100) NOT NULL,

	-- El monto lo pondré en numeric porque es muy buen formato para manejar el dinero, el 15
	-- hace referencia a los digitos de una cantidad y el dos es la escala (donde se pone el punto).
	monto			NUMERIC(15, 2) NOT NULL,
	fecha 			DATE NOT NULL,
	estatus			VARCHAR(20) NOT NULL DEFAULT 'normal' CHECK (estatus IN('normal', 'en_revision', 'sospechosa')),
	genera_alerta 	BOOLEAN NOT NULL DEFAULT false
);

-- Alertas !!

CREATE TABLE alerta(
	IDAlerta		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDCliente		UUID REFERENCES cliente(IDCliente),
	IDOperacion  	UUID NOT NULL REFERENCES operacion(IDOperacion),
	asignada_a 		UUID REFERENCES usuario(IDUsuario),
	resuelta_por 	UUID REFERENCES usuario(IDUsuario),
	motivo 			VARCHAR(300) NOT NULL,
	regla_rota		VARCHAR(150),

	prioridad		VARCHAR(10) NOT NULL CHECK (prioridad IN ('normal' , 'en_revision', 'sospechosa'))
	estatus			VARCHAR(20) NOT NULL DEFAULT 'pendiente' CHECK (alerta IN ('pendiente' , 'en_revision', 'investigando', 'cerrada', 'falso_positivo'))
	generada_en		TIMESTAMPTZ NOT NULL DEFAULT now(),
	resuelta_en 	TIMESTAMPTZ,
	resolucion		TEXT
);

-- lista de riesgo (versiones)
CREATE TABLE lista_riesgo(
	IDListaR		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	cargada_por		UUID REFERENCES usuario(IDUsuario),
	tipo			VARCHAR(5)NOT NULL CHECK (tipo IN ('PEP' , 'LPB'))
	versio			VARCHAR(20) NOT NULL,
	fecha_carga		TIMESTAMPTZ NOT NULL DEFAULT now(),
	cargada_por		UUID REFERENCES usuario(IDUsuario),
	activa			BOOLEAN NOT NULL DEFAULT true
);


-- Entradas INDIVIDUALES de cada lista
CREATE TABLE entrada_lista (
	IDEntradaL 		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDUsuario		UUID NOT NULL REFERENCES usuario(IDUsuario),
	nombre			VARCHAR(100) NOT NULL,
	identificado 	VARCHAR (50),
	pais			VARCHAR (20) NOT NULL,
	fecha_inclusion DATE
);

-- Validaciones de los clientes contra las listas
CREATE TABLE validacion_lista (
	IDValidacion	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	IDCliente		UUID NOT NULL REFERENCES cliente(IDCliente),
	IDListaR		UUID NOT NULL REFERENCES lista_riesgo(IDListaR),
	revisada_por	UUID REFERENCES usuario(IDUsuario),

	-- que onda
	score_coinci	NUMERIC(4,3),

	resultado 		VARCHAR(30) NOT NULL DEFAULT 'sin_coincidencia' CHECK (resultado IN('sin_coincidencia', 'posible_coincidencia', 'coincidencia_confirmada')),
	fecha			TIMESTAMPTZ NOT NULL DEFAULT now(),
	decision		TEXT
);

-- Reportes regulatorios
CREATE TABLE reporte_regulatorio (
	IDReporteR		UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	generado_por	UUID NOT NULL REFERENCES usuario(IDUsuario),
	autorizado_por	UUID NOT NULL REFERENCES usuario(IDUsuario),
	tipo_reporte	VARCHAR(15) NOT NULL,
	formato			VARCHAR(5) NOT NULL CHECK (formato IN ('XML' , 'txt')),
	ruta_archivo 	VARCHAR(500),
	acuse_recibo	VARCHAR(100),
	estatus			VARCHAR(20) NOT NULL DEFAULT 'generado' CHECK (estatus IN ('generado', 'en_revision', 'autorizado', 'enviado', 'observaciones')),
	generado_En		TIMESTAMPTZ NOT NULL DEFAULT now(),
	enviado_en 		TIMESTAMPTZ
);

-- Canal interno de reportes
CREATE TABLE reporte_interno (
	IDReporteInt	UUID PRIMARY KEY DEFAULT gen_random_uuid(),
	asignado_a		UUID REFERENCES usuario(IDUsuario),
	descripcion		TEXT NOT NULL,
	anonimo			BOOLEAN NOT NULL DEFAULT 'true',
	ruta_evidencia	VARCHAR(500),
	estatus			VARCHAR(20) NOT NULL CHECK (estatus IN ('recibido', 'en_revision' , 'investigando' , 'cerrado' )),
	fecha 			TIMESTAMPTZ NOT NULL DEFAULT now(),
	resolucion		TEXT
);

	entidad_afect	VARCHAR(50),

-- ================================

-- extension para usar la funcion uuid

CREATE EXTENSION IF NOT EXISTS "pgcrypto";


-- ================================
-- TABLA: usuario
-- ================================

CREATE TABLE usuario (
    IDUsuario       UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre          VARCHAR(200)    NOT NULL,
    email           VARCHAR(200)    NOT NULL UNIQUE,
    password_hash   VARCHAR(255)    NOT NULL,

    -- v2.0: se agregan los roles 'cliente' y 'admin'
    -- admin    = configura el sistema, gestiona usuario1s
    -- oficial  = cumplimiento PLD, acceso total al módulo
    -- operador = captura del día a día
    -- auditor  = solo lectura, visita 1-3 veces al año
    -- cliente  = portal de autoservicio para subir documentos
    rol             VARCHAR(20)     NOT NULL CHECK (rol IN ('admin', 'oficial', 'operador', 'auditor', 'cliente')),

    activo          BOOLEAN         NOT NULL DEFAULT true,
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT now()
    -- ..TZ es time zone
);


-- ================================
-- TABLA: cliente
-- ================================

-- ¿QUE PASO CON LOS NOT NULL?
-- hay campos que no cuentan con el not null por la razón de los clientes registrados,
-- sería imposible registrar datos hasta obtener algunos de los registros como curp hasta obtener ese dato
-- de igual forma hay clientes que no tienen domicilio físico en México o curp mexicana

-- ¿QUE PASO CON LOS ENUM's?
-- Si el cliente SVA LAW quisiera agregar un nuevo parámetro o estatus (imaginemos un nuevo rol), tendríamos
-- que bloquear temporalmente la tabla completa para cambiarlo. Por esto aquí usaremos VARCHAR + CHECK CONSTRAINT
-- permitiéndonos simplemente modificar la restricción sin tocar la estructura de la tabla.

CREATE TABLE cliente (
    IDCliente           UUID            PRIMARY KEY DEFAULT gen_random_uuid(),

    -- v2.0: FK al usuario del portal. Cada cliente tiene su propio login con rol='cliente'
    IDUsuario           UUID            REFERENCES usuario(IDUsuario),

    nombre_cliente      VARCHAR(255)    NOT NULL,
    RFC                 VARCHAR(13)     NOT NULL UNIQUE,
    curp                VARCHAR(18),
    domicilio           TEXT,

    -- v2.0: separamos en email personal e institucional
    email               VARCHAR(200),
    email_institucional VARCHAR(200),

    telefono            VARCHAR(20),

    -- aquí estamos creando lo que decía de los enums, pero con diferentes estados para elegir
    tipo_persona        VARCHAR(10)     NOT NULL CHECK (tipo_persona IN ('fisica', 'moral')),
    estatus             VARCHAR(10)     NOT NULL DEFAULT 'prospecto' CHECK (estatus IN ('prospecto', 'activo', 'suspendido')),
    nivel_riesgo        VARCHAR(10)     NOT NULL DEFAULT 'bajo' CHECK (nivel_riesgo IN ('bajo', 'medio', 'alto')),

    -- v2.0: score numérico dinámico calculado por el sistema (0.000 a 1.000)
    -- el nivel_riesgo es la etiqueta legible derivada de este score
    -- ejemplo: score=0.85 → nivel_riesgo='alto'
    score_riesgo        NUMERIC(4,3)    CHECK (score_riesgo >= 0 AND score_riesgo <= 1),

    bloqueado           BOOLEAN         NOT NULL DEFAULT false,
    created_at          TIMESTAMPTZ     NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: expediente
-- (1 a 1 con cliente)
-- ================================

CREATE TABLE expediente (
    IDExpediente    UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente       UUID        NOT NULL UNIQUE REFERENCES cliente(IDCliente),
    IDValidadoPor   UUID        REFERENCES usuario(IDUsuario),
    completo        BOOLEAN     NOT NULL DEFAULT false,
    activo          BOOLEAN     NOT NULL DEFAULT false,
    fecha_validac   TIMESTAMPTZ,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: documento
-- ================================

-- v2.0: se agregan formato y hash_integridad para cumplir NOM-104
-- NOM-104 obliga a guardar documentos en formatos que garanticen
-- integridad a largo plazo (PDF/A) y durante mínimo 10 años

CREATE TABLE documento (
    IDDocumento     UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    IDExpediente    UUID            NOT NULL REFERENCES expediente(IDExpediente),

    -- puede ser operador (rol=operador) o el propio cliente (rol=cliente)
    IDCargadoPor    UUID            REFERENCES usuario(IDUsuario),

    tipo_documento  VARCHAR(50)     NOT NULL,
    ruta_archivo    VARCHAR(500)    NOT NULL,

    -- v2.0: tipo MIME del archivo: 'application/pdf', 'image/jpeg', 'image/png'
    formato         VARCHAR(20),

    -- v2.0: hash SHA-256 del archivo para detectar si fue alterado (NOM-104)
    hash_integridad VARCHAR(64),

    vigente         BOOLEAN         NOT NULL DEFAULT true,
    fecha_carga     TIMESTAMPTZ     NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: modificacion_expediente
-- ================================

-- Historial de modificaciones al expediente
-- Esta tabla es INMUTABLE: sin UPDATE ni DELETE
-- Cualquier cambio al expediente genera un registro aquí

CREATE TABLE modificacion_expediente (
    IDModificacion  UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    IDExpediente    UUID            NOT NULL REFERENCES expediente(IDExpediente),
    IDUsuario       UUID            NOT NULL REFERENCES usuario(IDUsuario),
    campo_modif     VARCHAR(100)    NOT NULL,
    valor_anterior  TEXT,
    valor_nuevo     TEXT,

    -- justificacion es OBLIGATORIA. Validar en backend antes del INSERT
    justificacion   TEXT            NOT NULL,

    fecha           TIMESTAMPTZ     NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: contrato
-- ================================

-- v2.0: tipo_producto acotado a los dos productos definidos por el socio
-- credito_simple y credito_cuenta_corriente
-- los demás productos (factoraje, disposición) se escalan en versiones futuras

CREATE TABLE contrato (
    IDContrato      UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente       UUID            NOT NULL REFERENCES cliente(IDCliente),

    tipo_producto   VARCHAR(50)     NOT NULL CHECK (tipo_producto IN (
                                        'credito_simple',
                                        'credito_cuenta_corriente'
                                    )),

    -- El monto lo pondré en numeric porque es muy buen formato para manejar el dinero, el 15
    -- hace referencia a los dígitos de una cantidad y el dos es la escala (donde se pone el punto).
    monto           NUMERIC(15,2)   NOT NULL CHECK (monto > 0),

    fecha_inicio    DATE            NOT NULL,
    fecha_fin       DATE            NOT NULL,
    estatus         VARCHAR(20)     NOT NULL DEFAULT 'vigente' CHECK (estatus IN ('vigente', 'vencido', 'cancelado', 'en_revision')),

    CONSTRAINT contrato_fechas_check CHECK (fecha_fin > fecha_inicio)
);


-- ================================
-- TABLA: importacion_csv
-- ================================

-- v2.0: TABLA NUEVA
-- Registra cada lote de transacciones importado desde archivo CSV
-- El proceso batch nocturno crea un registro aquí antes de procesar las operaciones
-- Todas las operaciones de ese lote apuntan a este registro via IDImportacion

CREATE TABLE importacion_csv (
    IDImportacion       UUID        PRIMARY KEY DEFAULT gen_random_uuid(),

    -- null si fue ejecutado por proceso automático nocturno
    IDImportadoPor      UUID        REFERENCES usuario(IDUsuario),

    nombre_archivo      VARCHAR(300) NOT NULL,
    total_registros     INTEGER     NOT NULL DEFAULT 0,
    registros_ok        INTEGER     NOT NULL DEFAULT 0,
    registros_error     INTEGER     NOT NULL DEFAULT 0,

    estatus             VARCHAR(20) NOT NULL DEFAULT 'procesando' CHECK (estatus IN (
                                        'procesando',
                                        'completado',
                                        'parcial',
                                        'fallido'
                                    )),

    fecha_importacion   TIMESTAMPTZ NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: operacion
-- ================================

-- Operaciones financieras
-- Esta tabla es INMUTABLE: sin UPDATE ni DELETE una vez registrada
-- Para corregir un error se crea una operación de ajuste

CREATE TABLE operacion (
    IDOperacion     UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente       UUID            NOT NULL REFERENCES cliente(IDCliente),
    IDContrato      UUID            NOT NULL REFERENCES contrato(IDContrato),

    -- null si fue importada por CSV automático
    IDRegistradaPor UUID            REFERENCES usuario(IDUsuario),

    -- v2.0: FK al lote CSV de origen. null si fue capturada manualmente
    IDImportacion   UUID            REFERENCES importacion_csv(IDImportacion),

    tipo_operacion  VARCHAR(50)     NOT NULL CHECK (tipo_operacion IN ('desembolso', 'pago', 'disposicion')),

    -- El monto lo pondré en numeric porque es muy buen formato para manejar el dinero, el 15
    -- hace referencia a los dígitos de una cantidad y el dos es la escala (donde se pone el punto).
    monto           NUMERIC(15,2)   NOT NULL CHECK (monto > 0),

    fecha           DATE            NOT NULL,

    -- v2.0: para detección geográfica de cargos inusuales
    -- si el cliente siempre opera desde Querétaro y llega una op desde China → alerta
    ip_origen       VARCHAR(45),
    pais_origen     VARCHAR(100),

    estatus         VARCHAR(20)     NOT NULL DEFAULT 'normal' CHECK (estatus IN ('normal', 'en_revision', 'sospechosa')),
    genera_alerta   BOOLEAN         NOT NULL DEFAULT false
);


-- ================================
-- TABLA: tarjeta_cliente
-- ================================

-- v2.0: TABLA NUEVA
-- ADVERTENCIA PCI-DSS: nunca guardar número completo, CVV ni fecha de vencimiento
-- Solo se guardan los últimos 4 dígitos y un token del procesador de pagos
-- Esta tabla es para monitoreo PLD, no para procesar pagos

CREATE TABLE tarjeta_cliente (
    IDTarjeta           UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente           UUID        NOT NULL REFERENCES cliente(IDCliente),

    -- único dato visible del número de tarjeta
    ultimos_cuatro      VARCHAR(4)  NOT NULL CHECK (ultimos_cuatro ~ '^\d{4}$'),

    -- token generado por procesador certificado PCI-DSS (Stripe, Conekta, OpenPay)
    token_referencia    VARCHAR(200) NOT NULL,

    banco_emisor        VARCHAR(100),
    tipo_tarjeta        VARCHAR(10) NOT NULL CHECK (tipo_tarjeta IN ('credito', 'debito')),
    activa              BOOLEAN     NOT NULL DEFAULT true,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);


-- ================================
-- TABLA: alerta
-- ================================

-- Alertas !!
-- Esta tabla es INMUTABLE: sin DELETE
-- Solo el Oficial de Cumplimiento puede cambiar el estatus a 'cerrada'

CREATE TABLE alerta (
    IDAlerta        UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente       UUID        NOT NULL REFERENCES cliente(IDCliente),

    -- NULL si la alerta viene de una validación PEP/LPB, no de una operación
    -- por eso NO tiene NOT NULL: una alerta de lista no tiene operación asociada
    IDOperacion     UUID        REFERENCES operacion(IDOperacion),

    IDAsignadaA     UUID        REFERENCES usuario(IDUsuario),
    IDResueltaPor   UUID        REFERENCES usuario(IDUsuario),
    motivo          VARCHAR(300) NOT NULL,
    regla_rota      VARCHAR(150),

    -- alta/media/baja: qué tan urgente es atenderla
    prioridad       VARCHAR(10) NOT NULL CHECK (prioridad IN ('alta', 'media', 'baja')),

    -- pendiente → en_revision → investigando → cerrada o falso_positivo
    estatus         VARCHAR(20) NOT NULL DEFAULT 'pendiente' CHECK (estatus IN (
                                    'pendiente',
                                    'en_revision',
                                    'investigando',
                                    'cerrada',
                                    'falso_positivo'
                                )),

    generada_en     TIMESTAMPTZ NOT NULL DEFAULT now(),
    resuelta_en     TIMESTAMPTZ,
    resolucion      TEXT
);


-- ================================
-- TABLA: lista_riesgo
-- ================================

-- lista de riesgo (versiones)
-- Se mantienen versiones históricas: cuando se carga una nueva lista,
-- la anterior queda con activa=false pero NO se elimina

CREATE TABLE lista_riesgo (
    IDListaR        UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    tipo            VARCHAR(5)  NOT NULL CHECK (tipo IN ('PEP', 'LPB')),
    version         VARCHAR(20) NOT NULL,
    fecha_carga     TIMESTAMPTZ NOT NULL DEFAULT now(),

    -- solo el Oficial de Cumplimiento puede cargar listas
    IDCargadaPor    UUID        REFERENCES usuario(IDUsuario),

    -- solo true para la versión más reciente de cada tipo
    activa          BOOLEAN     NOT NULL DEFAULT true
);


-- ================================
-- TABLA: entrada_lista
-- ================================

-- Entradas INDIVIDUALES de cada lista
-- Cada fila es una persona dentro de una versión específica de lista

CREATE TABLE entrada_lista (
    IDEntradaL      UUID        PRIMARY KEY DEFAULT gen_random_uuid(),

    -- pertenece a una versión de lista_riesgo, NO a un usuario
    IDListaR        UUID        NOT NULL REFERENCES lista_riesgo(IDListaR),

    -- campo principal para el algoritmo de comparación por similitud
    nombre          VARCHAR(300) NOT NULL,

    -- RFC, CURP u otro identificador cuando está disponible en la lista oficial
    identificador   VARCHAR(50),

    pais            VARCHAR(100),
    fecha_inclusion DATE
);


-- ================================
-- TABLA: validacion_lista
-- ================================

-- Validaciones de los clientes contra las listas
-- Registra CADA validación, independientemente del resultado
-- Historial inmutable: evidencia ante la CNBV de que se está monitoreando

CREATE TABLE validacion_lista (
    IDValidacion    UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDCliente       UUID        NOT NULL REFERENCES cliente(IDCliente),
    IDListaR        UUID        NOT NULL REFERENCES lista_riesgo(IDListaR),
    IDRevisadaPor   UUID        REFERENCES usuario(IDUsuario),

    -- NUMERIC(4,3): 4 dígitos en total, 3 después del punto
    -- rango exacto de 0.000 a 1.000 sin errores de redondeo
    -- 0.000 = sin similitud | 1.000 = coincidencia exacta
    -- el umbral que genera alerta lo configura el Oficial
    score_coinci    NUMERIC(4,3) CHECK (score_coinci >= 0 AND score_coinci <= 1),

    resultado       VARCHAR(30) NOT NULL DEFAULT 'sin_coincidencia' CHECK (resultado IN (
                                    'sin_coincidencia',
                                    'posible_coincidencia',
                                    'coincidencia_confirmada'
                                )),

    fecha           TIMESTAMPTZ NOT NULL DEFAULT now(),
    decision        TEXT
);


-- ================================
-- TABLA: reporte_regulatorio
-- ================================

-- Reportes regulatorios a la CNBV
-- INMUTABLES una vez enviados: no se sobreescriben, se hacen reportes complementarios

CREATE TABLE reporte_regulatorio (
    IDReporteR      UUID        PRIMARY KEY DEFAULT gen_random_uuid(),

    -- puede ser null si el sistema lo genera automáticamente
    IDGeneradoPor   UUID        REFERENCES usuario(IDUsuario),

    -- solo el Oficial puede autorizar el envío
    IDAutorizadoPor UUID        REFERENCES usuario(IDUsuario),

    tipo_reporte    VARCHAR(50) NOT NULL,

    -- formato XML o TXT según lo exige la CNBV para cada tipo de reporte
    formato         VARCHAR(5)  NOT NULL CHECK (formato IN ('XML', 'TXT')),

    ruta_archivo    VARCHAR(500),

    -- código de confirmación recibido de la CNBV tras el envío exitoso
    acuse_recibo    VARCHAR(100),

    estatus         VARCHAR(20) NOT NULL DEFAULT 'generado' CHECK (estatus IN (
                                    'generado',
                                    'en_revision',
                                    'autorizado',
                                    'enviado',
                                    'con_observaciones'
                                )),

    generado_en     TIMESTAMPTZ NOT NULL DEFAULT now(),
    enviado_en      TIMESTAMPTZ
);


-- ================================
-- TABLA: reporte_interno
-- ================================

-- Canal interno de reportes
-- Canal confidencial de denuncias de empleados
-- Si anonimo=true el frontend NO envía IDUsuario al backend
-- el remitente nunca se guarda en ningún campo

CREATE TABLE reporte_interno (
    IDReporteInt    UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDAsignadoA     UUID        REFERENCES usuario(IDUsuario),
    descripcion     TEXT        NOT NULL,

    -- false por defecto: el usuario elige hacerlo anónimo
    -- si true: el sistema NO guarda quién lo envió bajo ninguna circunstancia
    anonimo         BOOLEAN     NOT NULL DEFAULT false,

    ruta_evidencia  VARCHAR(500),

    estatus         VARCHAR(20) NOT NULL DEFAULT 'recibido' CHECK (estatus IN (
                                    'recibido',
                                    'en_revision',
                                    'investigando',
                                    'cerrado',
                                    'escalado'
                                )),

    fecha           TIMESTAMPTZ NOT NULL DEFAULT now(),
    resolucion      TEXT
);


-- ================================
-- TABLA: bitacora
-- ================================

-- Bitácora (logs del sistema)
-- INMUTABLE: sin UPDATE ni DELETE bajo ninguna circunstancia
-- Registra todas las acciones de todos los roles incluyendo el nuevo rol 'cliente'

CREATE TABLE bitacora (
    IDBitacora      UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    IDUsuario       UUID        REFERENCES usuario(IDUsuario),
    accion          VARCHAR(200) NOT NULL,
    entidad_afect   VARCHAR(50),

    -- sin FK explícita porque puede apuntar a cualquier tabla del sistema
    id_entidad      UUID,

    ip_origen       VARCHAR(45),
    fecha           TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- al ejecutarse deberian verse las tablas en esto

SELECT
    table_name      AS tabla,
    (
        SELECT COUNT(*)
        FROM information_schema.columns c
        WHERE c.table_name = t.table_name
        AND c.table_schema = 'public'
    )               AS columnas
FROM information_schema.tables t
WHERE table_schema = 'public'
AND table_type = 'BASE TABLE'
ORDER BY table_name;

-- agregar nuevas columnas
ALTER TABLE cliente
    ADD COLUMN nombre           VARCHAR(100) NOT NULL DEFAULT '',
    ADD COLUMN apellido_paterno VARCHAR(100) NOT NULL DEFAULT '',
    ADD COLUMN apellido_materno VARCHAR(100),
    ADD COLUMN calle            VARCHAR(200),
    ADD COLUMN colonia          VARCHAR(100),
    ADD COLUMN municipio        VARCHAR(100),
    ADD COLUMN estado           VARCHAR(100),
    ADD COLUMN cp               VARCHAR(10),
    ADD COLUMN pais             VARCHAR(100) NOT NULL DEFAULT 'México';


-- para no cagotear todo lo que tenemos ...
-- migrar datos existentes (reparte nombre_cliente en las tres columnas)
UPDATE cliente SET
    nombre = SPLIT_PART(TRIM(nombre_cliente), ' ', 1),
    apellido_paterno = CASE
        WHEN ARRAY_LENGTH(STRING_TO_ARRAY(TRIM(nombre_cliente), ' '), 1) >= 2
        THEN SPLIT_PART(TRIM(nombre_cliente), ' ', 2)
        ELSE ''
    END,
    apellido_materno = CASE
        WHEN ARRAY_LENGTH(STRING_TO_ARRAY(TRIM(nombre_cliente), ' '), 1) >= 3
        THEN SPLIT_PART(TRIM(nombre_cliente), ' ', 3)
        ELSE NULL
    END,
    calle = domicilio
WHERE nombre_cliente IS NOT NULL;

ALTER TABLE cliente
    ALTER COLUMN nombre           DROP DEFAULT,
    ALTER COLUMN apellido_paterno DROP DEFAULT;

ALTER TABLE cliente
    DROP COLUMN nombre_cliente,
